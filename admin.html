function meetingStartDate(date, startTime) { if ( !/^\d{4}-\d{2}-\d{2}$/.test(date || "") || !/^\d{2}:\d{2}$/.test(startTime || "") ) { return null; } const value = new Date( `${date}T${startTime}:00+09:00` ); return Number.isNaN(value.getTime()) ? null : value; } async function loadMeetingSettings() { if (!sessionStorage.getItem("admin_access_token")) { return; } hide(ui.meetingStatus); try { const { response, data } = await workerPost( "/meeting-settings", {}, sessionStorage.getItem("admin_access_token") ); if (!response.ok) { throw new Error( data.error || "次回まち会議の設定を取得できませんでした。" ); } const meeting = data.meeting; if (!meeting) { ui.meetingDate.value = ""; ui.meetingStart.value = ""; ui.meetingEnd.value = ""; ui.meetingPlace.value = ""; ui.meetingNote.value = ""; show( ui.meetingStatus, "現在、次回まち会議は未設定です。" ); return; } ui.meetingDate.value = meeting.date || ""; ui.meetingStart.value = meeting.startTime || ""; ui.meetingEnd.value = meeting.endTime || ""; ui.meetingPlace.value = meeting.place || ""; ui.meetingNote.value = meeting.note || ""; const startsAt = meetingStartDate( meeting.date, meeting.startTime ); if ( startsAt && Date.now() >= startsAt.getTime() ) { show( ui.meetingStatus, "この開催日時はすでに過ぎているため、公開ページでは「未定」と表示されています。新しい予定を保存すると上書き更新されます。" ); } } catch (error) { show( ui.meetingStatus, error.message || "次回まち会議の設定を取得できませんでした。" ); } } ui.meetingForm.addEventListener( "submit", async event => { event.preventDefault(); if (isDummy()) { return; } hide(ui.meetingStatus); ui.meetingSave.disabled = true; try { const token = sessionStorage.getItem("admin_access_token") || ""; if (!token) { throw new Error( "ログイン期限が切れています。もう一度ログインしてください。" ); } const { response, data } = await workerPost( "/meeting-settings-save", { date: ui.meetingDate.value, startTime: ui.meetingStart.value, endTime: ui.meetingEnd.value, place: ui.meetingPlace.value.trim(), note: ui.meetingNote.value.trim(), }, token ); if (!response.ok) { throw new Error( data.error || "次回まち会議の設定を保存できませんでした。" ); } show( ui.meetingStatus, "次回まち会議の設定を更新しました。公開ページにも反映されます。" ); } catch (error) { show( ui.meetingStatus, error.message || "次回まち会議の設定を保存できませんでした。" ); } finally { ui.meetingSave.disabled = false; } } );
